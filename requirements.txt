PyQt6>=6.6                   # UI 框架
pyqtdarktheme-fork>=2.3      # 浅色/深色主题（MIT）—— 替代手写 QSS
                             # ⚠️ 必须装 -fork 版！PyPI 上的 `pyqtdarktheme`
                             #    是 Qt5 时代的老项目（0.1.7），没有
                             #    `setup_theme`，装上会 ImportError。
                             #    fork 版 import 名仍是 `qdarktheme`。
ldap3>=2.9.1                 # LDAP：查询 / 解锁 / 启用禁用 / 建对象 / 分页 / unicodePwd
pycryptodome>=3.20           # ⚠️ 硬依赖，不是可选优化 —— ldap3 的 NTLM 需要 MD4
                             #    ldap3 先找 Crypto.Hash.MD4（本包提供），找不到就回落
                             #    hashlib.new('MD4')；而 **Python 3.13 / OpenSSL 3 已把
                             #    MD4 移出内置库**（legacy provider）→ 回落那步抛
                             #    `ValueError: unsupported hash type MD4`。
                             #    表现：「绑定失败：发生未预期的错误（ValueError）」，
                             #    使用者完全无从下手；而**全部单元测试都用 fake ldap3，
                             #    从不真算 NTLM，所以一条都不会红**。
                             #    真域实测 2026-09-12（DC 192.0.2.10）：装上它之后
                             #    同一段代码立刻走到正确的「用户名或密码错误」。
pywin32>=306                 # LogonUser 身份模拟 + ADSI（RPC 通道改密）

pyinstaller>=6.0             # ⚠️ 打包必须用 .spec 而不是纯命令行 —— spec 里显式点名了
